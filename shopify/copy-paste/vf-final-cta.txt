{%- comment -%}
  Vettofit final CTA ("Start the ritual tonight.") — sections/vf-final-cta.liquid
  Built from "Vettofit Landing Page — Shopify Section Technical Guide", section 6 (Final CTA)
  and the Landing Page Guidelines, section 9. Quirky, but inside the brand book:
  flat shapes, tilted sticker chips, a polaroid pack card, and ONE playful line (the to-do list).
  No doodles, no exclamation marks.

  Buying:
  - Dog product (section setting) + optional "Cat version" block (or the dog product's
    vf.sibling_product). An in-section Dog / Cat switch also tells the rest of the page.
  - MRP (compare-at) crossed out, selling price, % off. Cost per day from product data.
  - One-time / Subscribe from Shopify selling plans. One-time is ALWAYS pre-selected.
  - Add to cart via AJAX, opens the cart drawer if there is one. Works without JavaScript too.
  Mobile: pack photo, heading, toggle, then a full-width button.
{%- endcomment -%}

{%- liquid
  assign ss = section.settings
  assign dog_product = ss.product
  assign cat_block = section.blocks | where: 'type', 'cat_product' | first
  assign cat_product = cat_block.settings.product
  if cat_product == blank and dog_product != blank
    assign cat_product = dog_product.metafields.vf.sibling_product.value
  endif
  comment
    Easiest option: find the cat product by its web address (handle). Accepts a full URL too.
  endcomment
  if cat_product == blank and ss.cat_handle != blank
    assign cat_handle = ss.cat_handle | split: '/products/' | last | split: '?' | first | split: '#' | first | remove: '/' | strip
    assign cat_product = all_products[cat_handle]
  endif
  assign has_cat = false
  if cat_product != blank and dog_product != blank and cat_product.id != dog_product.id
    assign has_cat = true
  endif
  assign heading_id = 'vf-final-h-' | append: section.id
-%}

    function formatMoney(cents, format) {
      var re = /\{\{\s*(\w+)\s*\}\}/, fmt = format || '₹{{amount}}';
      function d(n, p, t, dec) { t = t || ','; dec = dec || '.'; var parts = (n / 100).toFixed(p).split('.'); return parts[0].replace(/(\d)(?=(\d\d\d)+(?!\d))/g, '$1' + t) + (parts[1] ? dec + parts[1] : ''); }
      var m = fmt.match(re), v;
      switch (m ? m[1] : 'amount') {
        case 'amount_no_decimals': v = d(cents, 0); break;
        case 'amount_with_comma_separator': v = d(cents, 2, '.', ','); break;
        case 'amount_no_decimals_with_comma_separator': v = d(cents, 0, '.', ','); break;
        case 'amount_with_apostrophe_separator': v = d(cents, 2, "'", '.'); break;
        case 'amount_no_decimals_with_space_separator': v = d(cents, 0, ' '); break;
        case 'amount_with_space_separator': v = d(cents, 2, ' ', ','); break;
        case 'amount_with_period_and_space_separator': v = d(cents, 2, ' ', '.'); break;

      }
