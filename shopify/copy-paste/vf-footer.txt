{%- comment -%}
  Vettofit footer — sections/vf-footer.liquid (footer group)
  Built from "Vettofit Landing Page — Shopify Section Technical Guide", section 6 (Footer)
  and the Landing Page Guidelines, section 10.

  Quirky but on-brand: a teal Signal Line across the top with one coral drop, a slightly tilted
  newsletter card, and ONE playful line: the giant sign-off "See you at dinner." No doodles,
  no exclamation marks, and no trust-badge strip (proof sits next to its claim, never here).

  Blocks: Link column (max 4), Support (WhatsApp, email, hours, help link), Newsletter
  (Shopify customer form), Evidence and policies (menu), Social links (Theme settings → Social media).
  Mobile: link columns become accordions; Support, Newsletter and the market selector stay open.
  WhatsApp opens wa.me with a pre-filled message. No floating chat bubble.
  Locked: the footer logo version and the legal line format.
{%- endcomment -%}

<script>
  (function () {
    if (customElements.get('vf-footer')) return;
    class VfFooter extends HTMLElement {
      connectedCallback() {
        var self = this;
        this.desk = window.matchMedia('(min-width: 750px)');
        this.accs = Array.prototype.slice.call(this.querySelectorAll('[data-acc]'));

        /* Accordions on phones; always open (and not clickable) on desktop */
        this.sync = function () {
          self.accs.forEach(function (d) { d.open = self.desk.matches; });
        };
        this.accs.forEach(function (d) {
          d.querySelector('summary').addEventListener('click', function (e) { if (self.desk.matches) e.preventDefault(); });
        });
        this.sync();
        if (this.desk.addEventListener) this.desk.addEventListener('change', this.sync);

        /* Country selector submits on change */
        this.querySelectorAll('[data-autosubmit]').forEach(function (sel) {
          sel.addEventListener('change', function () { if (sel.form) sel.form.submit(); });
        });

        /* One coral drop drips from the Signal Line, once */
        var drop = this.querySelector('[data-drop]');
        if (drop && 'IntersectionObserver' in window && !window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
          this.io = new IntersectionObserver(function (en) {
            if (en[0].isIntersecting) { drop.classList.add('is-dripping'); self.io.disconnect(); }
          }, { threshold: 0.2 });
          this.io.observe(this);
        }

        /* Giant sign-off: always fits the footer width exactly */
        this.fitEl = this.querySelector('[data-fit]');
        this.fit = function () {
          var el = self.fitEl; if (!el) return;
          var box = el.parentElement;
          var cs = getComputedStyle(box);
          var avail = box.clientWidth - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight);
          el.style.fontSize = '100px';
          var w = el.scrollWidth;
          if (w > 0) el.style.fontSize = Math.min(100 * avail / w, 320).toFixed(2) + 'px';
        };
        this.fit();
        if (document.fonts && document.fonts.ready) document.fonts.ready.then(this.fit);
        this.onResize = function () { clearTimeout(self._ft); self._ft = setTimeout(self.fit, 80); };
        window.addEventListener('resize', this.onResize);

        /* Back to top */
        var top = this.querySelector('.vf-foot__top-link');
        if (top) top.addEventListener('click', function (e) {
          e.preventDefault();
          window.scrollTo({ top: 0, behavior: window.matchMedia('(prefers-reduced-motion: reduce)').matches ? 'auto' : 'smooth' });
        });
      }
      disconnectedCallback() {
        if (this.desk && this.desk.removeEventListener) this.desk.removeEventListener('change', this.sync);
        if (this.io) this.io.disconnect();
        window.removeEventListener('resize', this.onResize);
      }
    }
    customElements.define('vf-footer', VfFooter);
  })();
</script>
