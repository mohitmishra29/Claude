{%- comment -%}
  Vettofit pump story — sections/vf-pump-story.liquid  (v2)
  Built from "Vettofit Landing Page — Shopify Section Technical Guide", section 3 (Pump story),
  and "Landing Page Guidelines", moment 3 (The Pump).

  The scene: the real pack stands on White (product truth). The pump presses, one coral drop
  forms at the nozzle and falls straight down the teal Signal Line into a flat Carbon bowl.
  Everything is one SVG, so the line always starts exactly at the nozzle at any screen size.

  Desktop: pinned scene (about two screens of scroll). Scene left, steps right. Only the active
           step is open, so the column stays short and calm. If the scene can't fit the screen,
           or the theme blocks sticky, it falls back to still frames automatically.
  Mobile:  three swipeable frames (pump, drop, bowl) with dots. Tap the picture for the next one.
  Reduced motion: still frames side by side, no animation.

  All styles are scoped to this section's ID so theme heading styles can't resize the text.
{%- endcomment -%}

{%- for block in step_blocks -%}
  {%- liquid
    assign step_text = block.settings.text
    assign proof_label = ''
    assign proof_problem = ''
    if block.settings.proof != blank
      assign proof = block.settings.proof
      assign claim = proof.claim_id.value
      assign wording = proof.approved_wording | metafield_text
      assign expired = false
      if proof.expiry_date.value != blank
        assign exp_num = proof.expiry_date.value | date: '%Y%m%d' | plus: 0
        if exp_num < today_num
          assign expired = true
        endif
      endif
      if claim == blank
        assign proof_problem = 'Step text hidden: its Proof entry has no Claim ID.'
      elsif expired
        assign proof_problem = 'Step text hidden: its Proof entry is past its expiry date.'
      elsif wording == blank
        assign proof_problem = 'Step text hidden: its Proof entry has no approved wording.'
      endif
      if proof_problem != ''
        assign step_text = ''
      else
        assign step_text = wording
        assign proof_label = proof.type.value
      endif
    endif
    assign num = forloop.index | prepend: '0'
  -%}

<script>
  (function () {
    if (customElements.get('vf-pump-story')) return;

    function clamp(n) { return n < 0 ? 0 : n > 1 ? 1 : n; }

    /* Draws the scene for progress p (0 → 1):
       press the pump → a drop forms at the nozzle → the Signal Line draws down →
       the drop falls → it lands, a ripple, and the food takes the coral sheen. */
    function setStage(stage, p) {
      if (!stage) return;
      if (!stage._els) {
        var line = stage.querySelector('[data-line]');
        stage._els = {
          pack: stage.querySelector('[data-pack]'),
          line: line,
          len: line && line.getTotalLength ? line.getTotalLength() : 0,
          drop: stage.querySelector('[data-drop]'),
          ripple: stage.querySelector('[data-ripple]'),
          sheen: stage.querySelector('[data-sheen]'),
          x: parseFloat(stage.dataset.x),
          y0: parseFloat(stage.dataset.y0),
          y1: parseFloat(stage.dataset.y1)
        };
      }
      var e = stage._els;
      var press = clamp((p - 0.02) / 0.18);
      var form = clamp((p - 0.16) / 0.14);
      var lineT = clamp((p - 0.1) / 0.45);
      var fall = clamp((p - 0.33) / 0.3);
      var land = clamp((p - 0.63) / 0.22);

      if (e.pack) e.pack.setAttribute('transform', 'translate(0 ' + (Math.sin(Math.PI * press) * 4).toFixed(2) + ')');

      if (e.line && e.len) {
        var w = stage.getBoundingClientRect().width;
        if (w > 0) e.line.style.strokeWidth = (800 / w).toFixed(3); /* stays 2px on screen */
        e.line.classList.add('is-js');
        e.line.style.strokeDasharray = e.len.toFixed(1);
        e.line.style.strokeDashoffset = (e.len * (1 - lineT)).toFixed(2);
      }

      if (e.drop) {
        var y = e.y0 + 6 + (e.y1 - e.y0 - 6) * fall * fall; /* gravity: speeds up as it falls */
        var scale = 0.3 + form * 0.7;
        e.drop.setAttribute('transform', 'translate(' + e.x + ' ' + y.toFixed(1) + ') scale(' + scale.toFixed(2) + ')');
        e.drop.style.opacity = p > 0.16 && fall < 1 ? 1 : 0;
      }
      if (e.ripple) {
        e.ripple.style.opacity = land > 0 && land < 1 ? (1 - land).toFixed(3) : 0;
        e.ripple.style.transform = 'scale(' + (0.5 + land * 1.8).toFixed(3) + ')';
      }
      if (e.sheen) e.sheen.style.opacity = (land * 0.9).toFixed(3);
    }

    class VfPumpStory extends HTMLElement {
      connectedCallback() {
        var self = this;
        this.mode = this.dataset.mode;
        this.n = Math.max(1, parseInt(this.dataset.steps, 10) || 3);
        this.desk = window.matchMedia('(min-width: 990px)');
        this.rm = window.matchMedia('(prefers-reduced-motion: reduce)');
        this.track = this.querySelector('.vf-pump__track');
        this.pin = this.querySelector('[data-pin]');
        this.grid = this.querySelector('[data-grid]');
        this.mainStage = this.querySelector('.vf-pump__visual [data-stage]');
        this.steps = Array.prototype.slice.call(this.querySelectorAll('[data-step]'));
        this.frames = Array.prototype.slice.call(this.querySelectorAll('[data-frame]'));
        this.framesEl = this.querySelector('[data-frames]');
        this.dots = Array.prototype.slice.call(this.querySelectorAll('[data-dot]'));
        this.photos = Array.prototype.slice.call(this.querySelectorAll('[data-photo]'));
        this.video = this.querySelector('.vf-pump__visual video');
        this.activeFrame = -1;
        this.forced = null;
        this.stickyBroken = false;
        this.ticking = false;

        this.h = {
          scroll: this.onScroll.bind(this),
          resize: function () { clearTimeout(self._rz); self._rz = setTimeout(function () { self.layout(); }, 150); },
          layout: this.layout.bind(this),
          species: this.onSpecies.bind(this),
          blockSelect: this.onBlockSelect.bind(this),
          blockDeselect: this.onBlockDeselect.bind(this)
        };

        /* Listen to scroll only while the section is near the screen. */
        this.io = new IntersectionObserver(function (entries) {
          var on = entries[0].isIntersecting;
          if (on && !self.listening) { window.addEventListener('scroll', self.h.scroll, { passive: true }); self.listening = true; self.onScroll(); }
          if (!on && self.listening) { window.removeEventListener('scroll', self.h.scroll); self.listening = false; }
          self.toggleVideo(on);
        }, { rootMargin: '200px 0px' });
        this.io.observe(this);

        window.addEventListener('resize', this.h.resize);
        [this.desk, this.rm].forEach(function (mq) { if (mq.addEventListener) mq.addEventListener('change', self.h.layout); });
        document.addEventListener('vf:species-change', this.h.species);
        document.addEventListener('shopify:block:select', this.h.blockSelect);
        document.addEventListener('shopify:block:deselect', this.h.blockDeselect);

        this.setupFrames();
        this.setupCta();
        this.applySpecies(this.initialSpecies());
        this.layout();
      }

      disconnectedCallback() {
        var self = this;
        if (this.io) this.io.disconnect();
        if (this.frameIo) this.frameIo.disconnect();
        window.removeEventListener('scroll', this.h.scroll);
        window.removeEventListener('resize', this.h.resize);
        [this.desk, this.rm].forEach(function (mq) { if (mq.removeEventListener) mq.removeEventListener('change', self.h.layout); });
        document.removeEventListener('vf:species-change', this.h.species);
        document.removeEventListener('shopify:block:select', this.h.blockSelect);
        document.removeEventListener('shopify:block:deselect', this.h.blockDeselect);
      }

      endOf(i) { return (i + 1) / this.n; }
      startOf(i) { return i / this.n; }

      layout() {
        var self = this;
        var pinned = this.desk.matches && !this.rm.matches && !this.stickyBroken;
        this.classList.toggle('is-pinned', pinned);

        /* If the scene + steps don't fit the screen, show still frames instead of cutting off. */
        if (pinned) {
          var room = this.pin.clientHeight - 80 - 32;
          if (this.grid.offsetHeight > room + 4) { pinned = false; this.classList.remove('is-pinned'); }
        }

        this.frames.forEach(function (f, i) {
          var st = f.querySelector('[data-stage]');
          if (st) { st._els = null; setStage(st, self.endOf(i)); }
        });
        if (this.mainStage) this.mainStage._els = null;
        if (pinned) this.update();
        else setStage(this.mainStage, 1);
      }

      onScroll() {
        if (this.ticking) return;
        this.ticking = true;
        var self = this;
        requestAnimationFrame(function () { self.ticking = false; self.update(); });
      }

      update() {
        if (!this.classList.contains('is-pinned')) return;
        var r = this.track.getBoundingClientRect();
        var vh = window.innerHeight;

        /* Some themes block "sticky". If the scene isn't holding still, use still frames. */
        if (r.top < -60 && r.bottom > vh + 60) {
          if (Math.abs(this.pin.getBoundingClientRect().top) > 4) { this.stickyBroken = true; this.layout(); return; }
        }
        var p = this.forced !== null ? this.endOf(this.forced) - 0.001 : clamp(-r.top / Math.max(1, r.height - vh));
        this.render(p);
      }

      render(p) {
        setStage(this.mainStage, p);
        var idx = Math.min(this.n - 1, Math.floor(p * this.n));
        this.steps.forEach(function (s, i) { s.classList.toggle('is-active', i === idx); });
        this.photos.forEach(function (ph) { ph.classList.toggle('is-on', parseInt(ph.dataset.photo, 10) === idx); });
      }

      /* ---------- Mobile frames ---------- */
      setupFrames() {
        var self = this;
        if (!this.framesEl || !this.frames.length) return;
        this.frameIo = new IntersectionObserver(function (entries) {
          entries.forEach(function (en) {
            if (en.isIntersecting && en.intersectionRatio >= 0.6) self.setFrame(parseInt(en.target.dataset.index, 10));
          });
        }, { root: this.framesEl, threshold: [0.6] });
        this.frames.forEach(function (f) { self.frameIo.observe(f); });

        this.dots.forEach(function (d) {
          d.addEventListener('click', function () { self.goTo(parseInt(d.dataset.index, 10)); });
        });
        this.querySelectorAll('[data-next]').forEach(function (m) {
          m.addEventListener('click', function () {
            if (self.desk.matches) return;
            self.goTo((Math.max(0, self.activeFrame) + 1) % self.frames.length);
          });
        });
      }

      goTo(i) {
        var f = this.frames[i];
        if (!f || !this.framesEl) return;
        var pad = parseFloat(getComputedStyle(this.framesEl).paddingLeft) || 0;
        var left = this.framesEl.scrollLeft + f.getBoundingClientRect().left - this.framesEl.getBoundingClientRect().left - pad;
        this.framesEl.scrollTo({ left: left, behavior: this.rm.matches ? 'auto' : 'smooth' });
      }

      setFrame(i) {
        if (i === this.activeFrame) return;
        this.activeFrame = i;
        this.dots.forEach(function (d, k) {
          d.classList.toggle('is-active', k === i);
          if (k === i) d.setAttribute('aria-current', 'true'); else d.removeAttribute('aria-current');
        });
        if (this.desk.matches || this.rm.matches) return;
        var stage = this.frames[i] && this.frames[i].querySelector('[data-stage]');
        if (!stage) return;
        var from = this.startOf(i), to = this.endOf(i), t0 = null, dur = 1100;
        if (stage._raf) cancelAnimationFrame(stage._raf);
        var tick = function (ts) {
          if (t0 === null) t0 = ts;
          var k = clamp((ts - t0) / dur);
          setStage(stage, from + (to - from) * k);
          if (k < 1) stage._raf = requestAnimationFrame(tick);
        };
        stage._raf = requestAnimationFrame(tick);
      }

      toggleVideo(on) {
        if (!this.video) return;
        if (on && !this.rm.matches) { var pr = this.video.play(); if (pr && pr.catch) pr.catch(function () {}); }
        else this.video.pause();
      }

      /* ---------- Dog / Cat ---------- */
      initialSpecies() {
        var s = null;
        try { s = new URLSearchParams(window.location.search).get('pet'); } catch (e) {}
        if (s !== 'dog' && s !== 'cat') { try { s = localStorage.getItem('vf_pet'); } catch (e) {} }
        return s === 'cat' ? 'cat' : 'dog';
      }
      onSpecies(e) { var d = (e && e.detail) || {}; this.applySpecies(d.species || d.pet || d.value); }
      applySpecies(species) {
        if (species !== 'dog' && species !== 'cat') return;
        this.querySelectorAll('[data-pack]').forEach(function (pack) {
          var imgs = pack.querySelectorAll('[data-species]');
          if (!imgs.length) return;
          var has = pack.querySelector('[data-species="' + species + '"]');
          var want = has ? species : 'dog';
          imgs.forEach(function (img) { img.classList.toggle('is-on', img.getAttribute('data-species') === want); });
        });
      }

      /* ---------- Button ---------- */
      setupCta() {
        var cta = this.querySelector('[data-cta]');
        if (!cta || cta.dataset.target !== 'dose') return;
        var check = function () {
          var ok = !!document.getElementById('vf-dose');
          var editor = window.Shopify && window.Shopify.designMode;
          cta.hidden = !ok && !editor;
        };
        if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', check);
        else check();
      }

      /* ---------- Theme editor: jump to the step you're editing ---------- */
      onBlockSelect(e) {
        var id = e.detail && e.detail.blockId;
        var step = id && this.querySelector('[data-step][data-block-id="' + id + '"]');
        if (!step) return;
        var i = parseInt(step.dataset.index, 10);
        if (this.classList.contains('is-pinned')) { this.forced = i; this.render(this.endOf(i) - 0.001); }
        else this.goTo(i);
      }
      onBlockDeselect() { this.forced = null; this.onScroll(); }
    }

    customElements.define('vf-pump-story', VfPumpStory);
  })();
</script>
